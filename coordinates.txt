<script>

let map;

let routeLine;

let startMarker;

let endMarker;



function findRoute() {


    // Get values

    const startLat =
        parseFloat(document.getElementById("startLat").value);

    const startLon =
        parseFloat(document.getElementById("startLon").value);

    const endLat =
        parseFloat(document.getElementById("endLat").value);

    const endLon =
        parseFloat(document.getElementById("endLon").value);

    const vehicle =
        document.getElementById("vehicle").value;



    // Validate

    if (
        isNaN(startLat) ||
        isNaN(startLon) ||
        isNaN(endLat) ||
        isNaN(endLon)
    ) {

        alert("Please enter all coordinates.");

        return;
    }


    if (vehicle === "") {

        alert("Please select Bike or Car.");

        return;
    }



    // Show map

    document.getElementById("map").style.display = "block";



    // Create map

    if (!map) {

        map = L.map("map");

        L.tileLayer(
            "https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png",
            {
                attribution:
                    "&copy; OpenStreetMap contributors"
            }
        ).addTo(map);

    }



    // Remove old route

    if (routeLine) {

        map.removeLayer(routeLine);

    }


    if (startMarker) {

        map.removeLayer(startMarker);

    }


    if (endMarker) {

        map.removeLayer(endMarker);

    }



    // -----------------------------------------
    // CREATE OSRM URL
    // -----------------------------------------

    let url =
        `https://router.project-osrm.org/route/v1/driving/` +
        `${startLon},${startLat};${endLon},${endLat}` +
        `?overview=full&geometries=geojson`;



    // -----------------------------------------
    // BIKE ROUTE
    // -----------------------------------------

    if (vehicle === "bike") {

        url =
            `https://router.project-osrm.org/route/v1/driving/` +
            `${startLon},${startLat};${endLon},${endLat}` +
            `?overview=full&geometries=geojson&exclude=motorway`;

    }



    document.getElementById("result").innerHTML =
        "Finding route...";



    // -----------------------------------------
    // CALL API
    // -----------------------------------------

    fetch(url)

        .then(response => response.json())

        .then(data => {


            if (data.code !== "Ok") {

                document.getElementById("result").innerHTML =
                    "❌ Route could not be found.";

                return;
            }


            const route = data.routes[0];



            // -----------------------------------------
            // ROUTE GEOMETRY
            // -----------------------------------------

            const coordinates =
                route.geometry.coordinates.map(
                    point => [point[1], point[0]]
                );



            // -----------------------------------------
            // DRAW ROUTE
            // -----------------------------------------

            routeLine = L.polyline(
                coordinates,
                {
                    weight: 6
                }
            ).addTo(map);



            // -----------------------------------------
            // START MARKER
            // -----------------------------------------

            startMarker =
                L.marker([startLat, startLon])
                .addTo(map)
                .bindPopup("📍 Starting Location");



            // -----------------------------------------
            // END MARKER
            // -----------------------------------------

            endMarker =
                L.marker([endLat, endLon])
                .addTo(map)
                .bindPopup("📍 Ending Location");



            // -----------------------------------------
            // FIT MAP
            // -----------------------------------------

            map.fitBounds(
                routeLine.getBounds()
            );



            // -----------------------------------------
            // DISTANCE
            // -----------------------------------------

            const distance =
                (route.distance / 1000).toFixed(2);



            // -----------------------------------------
            // TIME
            // -----------------------------------------

            const totalMinutes =
                Math.round(route.duration / 60);


            const hours =
                Math.floor(totalMinutes / 60);


            const minutes =
                totalMinutes % 60;


            let timeText;


            if (hours > 0) {

                timeText =
                    `${hours} hr ${minutes} min`;

            } else {

                timeText =
                    `${minutes} min`;

            }



            // -----------------------------------------
            // VEHICLE NAME
            // -----------------------------------------

            const vehicleName =
                vehicle === "bike"
                ? "🏍️ Bike"
                : "🚗 Car";



            // -----------------------------------------
            // RESULT
            // -----------------------------------------

            document.getElementById("result").innerHTML =

                `<strong>✅ Route Found</strong>

                <br>

                Vehicle:
                ${vehicleName}

                <br>

                Distance:
                ${distance} km

                <br>

                Estimated Time:
                ${timeText}`;


        })


        .catch(error => {

            console.error(error);

            document.getElementById("result").innerHTML =
                "❌ Unable to connect to routing service.";

        });

}

</script>
